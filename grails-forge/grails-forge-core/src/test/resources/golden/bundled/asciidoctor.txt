== gradle/asciidoc.gradle

+     version '3.0.0'
+     modules {
+         diagram {
+             version '2.3.1'
+         }
+     }
+ 
+     options doctype: "book", ruby: "erubis"
+ 
+     attributes "sourcedir": "src/docs/asciidoc",
+                "source-highlighter": "coderay",
+                "toc": "left",
+                "idprefix": "",
+                "idseparator": "-",
+                "icons": "font",
+                "setanchors": "",
+                "listing-caption": "",
+                "imagesdir": "images",
+                "project-version": "$project.version",
+                "revnumber": "$project.version"
+ }
+ 
+ // The Asciidoctor Gradle plugin does not support the configuration cache, so a build that runs
+ // its task runs without the cache instead of failing. This script cannot see the plugin's classes,
+ // so it names the task rather than its type.
+ tasks.named('asciidoctor') {
+     notCompatibleWithConfigurationCache('The Asciidoctor Gradle plugin does not support the configuration cache')
+ }
+
